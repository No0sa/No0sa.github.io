class QuranBloc extends Bloc<QuranEvent, QuranState> {
  final QuranRepository quranRepository;
  final SettingsRepository settingsRepository;

  QuranBloc({required this.quranRepository, required this.settingsRepository}) : super(QuranInitial()) {
    on<LoadAllSurahs>(_onLoadAllSurahs);
    on<LoadSurah>(_onLoadSurah);
    on<ToggleBookmark>(_onToggleBookmark);
    on<UpdateLastRead>(_onUpdateLastRead);
    on<LoadRandomAyah>(_onLoadRandomAyah);
  }

  Future<void> _onLoadSurah(LoadSurah event, Emitter<QuranState> emit) async {
    emit(QuranLoading());
    try {
      final surah = await quranRepository.getSurahByNumber(event.surahNumber);
      if (surah == null) { emit(const QuranError('Surah not found')); return; }
      final ayahs = await quranRepository.getAyahsBySurah(event.surahNumber);
      final bookmarks = await settingsRepository.getBookmarks();
      emit(SurahLoaded(surah, ayahs, bookmarks));
    } catch (e) { emit(QuranError('Failed to load Surah: ${e.toString()}')); }
  }

  Future<void> _onToggleBookmark(ToggleBookmark event, Emitter<QuranState> emit) async {
    if (state is SurahLoaded) {
      final currentState = state as SurahLoaded;
      final bookmarks = List<String>.from(currentState.bookmarks);
      final bookmark = '${event.surahNumber}:${event.ayahNumber}';
      if (bookmarks.contains(bookmark)) {
        await settingsRepository.removeBookmark(event.surahNumber, event.ayahNumber);
        bookmarks.remove(bookmark);
      } else {
        await settingsRepository.addBookmark(event.surahNumber, event.ayahNumber);
        bookmarks.add(bookmark);
      }
      emit(SurahLoaded(currentState.surah, currentState.ayahs, bookmarks));
    }
  }
}

class CemeteryBloc extends Bloc<CemeteryEvent, CemeteryState> {
  CemeteryBloc() : super(CemeteryInitial()) { on<LoadCemeteryData>(_onLoadData); }

  Future<void> _onLoadData(LoadCemeteryData event, Emitter<CemeteryState> emit) async {
    emit(CemeteryLoading());
    try {
      Position? position = await _getCurrentLocation();
      LatLng userLatLng = position != null
          ? LatLng(position.latitude, position.longitude)
          : const LatLng(25.276989, 55.296249);
      final jsonString = await rootBundle.loadString('assets/images/json/data.json');
      final List<dynamic> jsonData = json.decode(jsonString);
      Set<Marker> markers = {};
      markers.addAll(jsonData.map((item) {
        final lat = double.tryParse(item['latitude'].toString()) ?? 0.0;
        final lng = double.tryParse(item['longitude'].toString()) ?? 0.0;
        if (_isLocationInUAE(lat, lng)) {
          return Marker(markerId: MarkerId('$lat-$lng'), position: LatLng(lat, lng), infoWindow: InfoWindow(title: item['name'] ?? ''));
        }
        return null;
      }).whereType<Marker>());
      emit(CemeteryLoaded(markers: markers, initialPosition: userLatLng));
    } catch (e) { emit(CemeteryError(e.toString())); }
  }
  bool _isLocationInUAE(double lat, double lng) => lat >= 24.0 && lat <= 26.0 && lng >= 51.0 && lng <= 57.0;
}

class ChatService {
  static late IO.Socket socket;
  static bool isConnected = false;

  static void initSocket(String userId) {
    socket = IO.io('http://127.0.0.1:3000', <String, dynamic>{'transports': ['websocket'], 'autoConnect': true});
    socket.onConnect((_) { socket.emit('joinRoom', userId); isConnected = true; });
    socket.onDisconnect((_) => isConnected = false);
  }

  static void sendMessage(String receiverId, String message) {
    if (isConnected) socket.emit('sendMessage', {'receiverId': receiverId, 'message': message, 'timestamp': DateTime.now().toIso8601String()});
  }
  static void onMessageReceived(Function(Map<String, dynamic>) callback) { socket.on('receiveMessage', (data) => callback(data)); }
  static void joinRoom(String roomId) { if (isConnected) socket.emit('joinRoom', roomId); }
  static void dispose() => socket.dispose();
}

class ApiService {
  static const String baseUrl = 'http://127.0.0.1:8000/api';

  static Future<Map<String, String>> getHeaders() async {
    final prefs = await SharedPreferences.getInstance();
    final token = prefs.getString('token');
    return {'Content-Type': 'application/json', 'Accept': 'application/json', 'X-Requested-With': 'XMLHttpRequest', if (token != null) 'Authorization': 'Bearer $token'};
  }
  static Future<http.Response> bookAppointment(Map<String, dynamic> data) async => await http.post(Uri.parse('$baseUrl/appointments'), headers: await getHeaders(), body: json.encode(data));
  static Future<http.Response> getDoctorsBySpecialty(String specialtyId) async => await http.get(Uri.parse('$baseUrl/doctors?specialty=$specialtyId'), headers: await getHeaders());
  static Future<http.Response> processWalletPayment(Map<String, dynamic> data) async => await http.post(Uri.parse('$baseUrl/wallet/payment'), headers: await getHeaders(), body: json.encode(data));
  static Future<http.Response> getChatMessages(String chatId) async => await http.get(Uri.parse('$baseUrl/chats/$chatId/messages'), headers: await getHeaders());
}

class _MainContentState extends State<MainContent> {
  final FirebaseFirestore _firestore = FirebaseFirestore.instance;
  List<User> employees = [];
  String selectedDepartment = 'All';

  Future<void> fetchEmployees() async {
    final snapshot = await _firestore.collection('employees').get();
    setState(() {
      employees = snapshot.docs.map((doc) {
        final data = doc.data() as Map<String, dynamic>;
        return User(id: doc.id, name: data['name'] ?? 'Unknown', department: data['department'] ?? 'Unknown', designation: data['designation'] ?? 'Unknown', email: data['email'] ?? '', role: data['role'] ?? 'Employee');
      }).toList();
    });
  }

  List<User> getFilteredEmployees() => selectedDepartment == 'All' ? employees : employees.where((e) => e.department == selectedDepartment).toList();

  void deleteEmployee(String id) async {
    await _firestore.collection('employees').doc(id).delete();
    fetchEmployees();
  }
}

// Laravel admin dashboard — Raheel UAE Government cemetery management system
class RaheelDashboardController extends Controller
{
    public function index()
    {
        $totalBurials    = Burial::count();
        $totalCemeteries = Cemetery::count();
        $todayBurials    = Burial::whereDate('burial_date', today())->count();
        $pendingRequests = BurialRequest::where('status', 'pending')->count();
        $recentBurials   = Burial::with(['cemetery', 'requestedBy'])->latest()->take(10)->get();
        $cemeteryStats   = Cemetery::withCount('burials')->orderByDesc('burials_count')->get();

        return view('dashboard.index', compact('totalBurials', 'totalCemeteries', 'todayBurials', 'pendingRequests', 'recentBurials', 'cemeteryStats'));
    }

    public function apiStats()
    {
        return response()->json([
            'total_burials'     => Burial::count(),
            'total_cemeteries'  => Cemetery::count(),
            'today_burials'     => Burial::whereDate('burial_date', today())->count(),
            'pending_requests'  => BurialRequest::where('status', 'pending')->count(),
            'approved_requests' => BurialRequest::where('status', 'approved')->count(),
            'monthly_burials'   => Burial::whereMonth('burial_date', now()->month)->count(),
        ]);
    }

    // Paginated burial records with search, cemetery filter & date range — consumed by Flutter app
    public function burials(Request $request)
    {
        $query = Burial::with(['cemetery', 'requestedBy', 'approvedBy']);

        if ($request->cemetery_id)  $query->where('cemetery_id', $request->cemetery_id);
        if ($request->search)       $query->where('deceased_name', 'like', '%' . $request->search . '%');
        if ($request->date_from && $request->date_to)
            $query->whereBetween('burial_date', [$request->date_from, $request->date_to]);

        return response()->json($query->latest()->paginate(20));
    }

    // Approve request and create burial record atomically
    public function approveRequest(BurialRequest $burialRequest)
    {
        DB::transaction(function () use ($burialRequest) {
            $burialRequest->update(['status' => 'approved', 'approved_by' => auth()->id(), 'approved_at' => now()]);
            Burial::create([
                'cemetery_id'     => $burialRequest->cemetery_id,
                'deceased_name'   => $burialRequest->deceased_name,
                'burial_date'     => $burialRequest->requested_date,
                'grave_latitude'  => $burialRequest->grave_latitude,
                'grave_longitude' => $burialRequest->grave_longitude,
                'requested_by'    => $burialRequest->user_id,
                'approved_by'     => auth()->id(),
            ]);
        });
        return response()->json(['message' => 'Request approved successfully']);
    }
}

void _startLiveLocationUpdates() {
  locationStream = Geolocator.getPositionStream(locationSettings: locationSettings).listen((Position position) {

      userLocation = position;
      markers.removeWhere((m) => m.markerId.value == 'userLocation');
      markers.add(Marker(markerId: MarkerId('userLocation'), position: LatLng(userLocation!.latitude, userLocation!.longitude), infoWindow: InfoWindow(title: 'Your Location', snippet: distanceToGrave != null ? 'Distance: ${distanceToGrave!.toStringAsFixed(2)}m' : 'Calculating...')));
      distanceToGrave = Geolocator.distanceBetween(userLocation!.latitude, userLocation!.longitude, widget.grave['latitude'], widget.grave['longitude']);
      _getPolylinePoints();
    });
  });
}

Future<void> _getPolylinePoints() async {
  PolylineResult result = await PolylinePoints().getRouteBetweenCoordinates(
    googleApiKey: apiKey,
    request: PolylineRequest(origin: PointLatLng(userLocation!.latitude, userLocation!.longitude), destination: PointLatLng(widget.grave['latitude'], widget.grave['longitude']), mode: TravelMode.walking),
  );
  if (result.points.isNotEmpty) {
    setState(() { polylines.add(Polyline(polylineId: PolylineId('poly'), color: Color(0xFFF47C20), points: result.points.map((p) => LatLng(p.latitude, p.longitude)).toList(), width: 8)); });
  }
}

class AthkarBloc extends Bloc<AthkarEvent, AthkarState> {
  final AthkarRepository athkarRepository;

  AthkarBloc({required this.athkarRepository}) : super(AthkarInitial()) {
    on<LoadAthkarCategories>(_onLoadAthkarCategories);
    on<LoadAthkarByCategory>(_onLoadAthkarByCategory);
  }

  Future<void> _onLoadAthkarCategories(LoadAthkarCategories event, Emitter<AthkarState> emit) async {
    emit(AthkarLoading());
    try { emit(AthkarCategoriesLoaded(await athkarRepository.getCategories())); }
    catch (e) { emit(AthkarError('Failed to load categories: ${e.toString()}')); }
  }

  Future<void> _onLoadAthkarByCategory(LoadAthkarByCategory event, Emitter<AthkarState> emit) async {
    emit(AthkarLoading());
    try { emit(AthkarLoaded(event.category, await athkarRepository.getAthkarByCategory(event.category))); }
    catch (e) { emit(AthkarError('Failed to load Athkar: ${e.toString()}')); }
  }
}
